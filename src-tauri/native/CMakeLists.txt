cmake_minimum_required(VERSION 3.20)
project(fetchrail_torrent LANGUAGES CXX)
find_package(LibtorrentRasterbar 2.1.2 EXACT CONFIG REQUIRED)
add_library(fetchrail_torrent STATIC torrent.cpp)
target_compile_features(fetchrail_torrent PRIVATE cxx_std_17)
target_include_directories(fetchrail_torrent PRIVATE vendor-cache)
target_link_libraries(fetchrail_torrent PRIVATE LibtorrentRasterbar::torrent-rasterbar)
if(MSVC)
  target_compile_options(fetchrail_torrent PRIVATE /EHsc /O2 /utf-8)
  set_property(TARGET fetchrail_torrent PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
endif()
install(TARGETS fetchrail_torrent ARCHIVE DESTINATION lib)
option(FETCHRAIL_BUILD_TORRENT_FIXTURE "Build the native-only comparison fixture" OFF)
if(FETCHRAIL_BUILD_TORRENT_FIXTURE)
  add_executable(torrent-native-fixture harness.cpp)
  target_compile_features(torrent-native-fixture PRIVATE cxx_std_17)
  target_include_directories(torrent-native-fixture PRIVATE vendor-cache)
  target_link_libraries(torrent-native-fixture PRIVATE fetchrail_torrent LibtorrentRasterbar::torrent-rasterbar)
  if(MSVC)
    set_property(TARGET torrent-native-fixture PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
  endif()
endif()
